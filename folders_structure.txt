GenomeAnalysis/
├── analyses/                  # Output from all bioinformatics pipelines
│   ├── 01_preprocessing/      # FastQC reports and Trimmomatic outputs for all reads
│   ├── 02_genome_assembly/    # Flye assembly (Nanopore) and YaHS Hi-C scaffolding
│   ├── 03_polishing/          # Pilon polished assembly using Illumina reads
│   ├── 04_assessment/         # Assembly evaluation using BUSCO and QUAST
│   ├── 05_annotation/         # Repeat masking, BRAKER3 structural, and EggNOG functional annotation
│   ├── 06_expression/         # STAR mapping, featureCounts, and DESeq2 differential expression
│   └── 07_chloroplast/        # GetOrganelle assembly and chloroplast read mapping
├── code/                      # All bash and R scripts used for the analyses, numbered sequentially
├── data/                      # Input data and metadata
│   ├── metadata/              # Sample information and experimental design 
│   ├── raw_data/              # Symlinks to read-only raw sequence files (Nanopore, Illumina, Hi-C, RNA-seq)
│   ├── reference_data/        # Protein reference datasets for annotation
│   └── trimmed_data/          # Cleaned and trimmed FASTQ files ready for downstream tools
├── data_for_IGV/              # Formatted files prepped for local visualization in IGV
├── figures/                   # Final generated plots and screenshots for reports/presentations
│   ├── DESeq/                 # Differential expression visualizations (PCA, Volcano, Heatmap)
│   ├── extra_analyses/        # Custom plots (Hi-C Juicebox, Chloroplast mapping)
│   ├── Flye/                  # Bandage assembly graphs
│   ├── IGV/                   # Genome browser screenshots
│   ├── illumina/              # FastQC visual summaries
│   ├── nanopore/              # Long-read quality distributions
│   └── QUAST/                 # Assembly quality metric plots
├── logs/                      # SLURM output logs (.out) and software execution records
├── README.md                  # Project overview and GitHub repository landing page
└── folders_structure.txt      # This directory map
